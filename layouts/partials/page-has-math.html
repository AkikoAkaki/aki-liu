{{- /* Front matter `math` wins; otherwise regular pages are scanned for math delimiters and list pages return false. */ -}} {{- $hasMath := false -}} {{- if isset .Params "math" -}} {{- $hasMath = .Params.math -}} {{- else if .IsPage -}} {{- $raw := .RawContent -}} {{- if or (in $raw "$") (in $raw "\\(") (in $raw "\\[") -}} {{- $hasMath = gt (len (findRE `\$\$[\s\S]*?\$\$|\\\([\s\S]*?\\\)|\\\[[\s\S]*?\\\]|(^|[^\\])\$[^$\n]+\$` $raw 1)) 0 -}} {{- end -}} {{- end -}} {{- return $hasMath -}}
