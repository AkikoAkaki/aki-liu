{{- /* Returns whether a page needs KaTeX. Front matter `enableKaTeX` or `math` wins (Hugo lowercases param keys); otherwise regular pages are scanned for math delimiters and list pages return false. */ -}} {{- $override := "" -}} {{- $hasOverride := false -}} {{- range slice "enablekatex" "math" -}} {{- if and (not $hasOverride) (isset $.Params .) -}} {{- $hasOverride = true -}} {{- $override = index $.Params . -}} {{- end -}} {{- end -}} {{- $hasMath := false -}} {{- if $hasOverride -}} {{- $hasMath = partial "is-truthy.html" $override -}} {{- else if .IsPage -}} {{- $raw := .RawContent -}} {{- if or (in $raw "$") (in $raw "\\(") (in $raw "\\[") -}} {{- $hasMath = gt (len (findRE `\$\$[\s\S]*?\$\$|\\\([\s\S]*?\\\)|\\\[[\s\S]*?\\\]|(^|[^\\])\$[^$\n]+\$` $raw 1)) 0 -}} {{- end -}} {{- end -}} {{- return $hasMath -}}
