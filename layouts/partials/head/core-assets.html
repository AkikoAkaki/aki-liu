{{- /* Static head assets cached across pages in one build. */ -}} {{- /* Fonts */ -}} {{- /* CSS pipeline: global base bundle + route-level styles */ -}} {{- $base := resources.Get "css/base.css" -}} {{- $layout := resources.Get "css/layout.css" -}} {{- $components := resources.Get "css/components.css" -}} {{- $search := resources.Get "css/search.css" -}} {{- $style := slice $base $layout $components $search | resources.Concat "css/bundle.css" | resources.Minify | fingerprint -}} {{- if .IsHome -}} {{- $home := resources.Get "css/home.css" -}} {{- $homeStyle := $home | resources.Minify | fingerprint -}} {{- end -}} {{- if .IsPage -}} {{- $post := resources.Get "css/post.css" -}} {{- $postStyle := $post | resources.Minify | fingerprint -}} {{- end -}} {{- if eq .Section "microblog" -}} {{- $microblog := resources.Get "css/microblog.css" -}} {{- $microblogStyle := $microblog | resources.Minify | fingerprint -}} {{- end -}} {{- $mainJS := resources.Get "js/main.js" | js.Build (dict "targetPath" "js/main.js") | resources.Minify | fingerprint -}} {{- if and (eq .Section "microblog") (eq .Kind "section") -}} {{- $microblogJS := resources.Get "js/microblog.js" | js.Build (dict "targetPath" "js/microblog.js") | resources.Minify | fingerprint -}} {{- end -}}
