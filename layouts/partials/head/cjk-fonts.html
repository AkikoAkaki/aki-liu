{{- /* Content-dependent, so it stays outside the cached core assets. Scans what a visitor can see here: homepage data, or the page text plus archive previews and tags, or every microblog entry. */ -}} {{- $needsCJK := and (eq .Language.Lang "zh") (not .IsHome) -}} {{- if not $needsCJK -}} {{- $text := "" -}} {{- if .IsHome -}} {{- $text = jsonify site.Data.homepage -}} {{- else -}} {{- $text = printf "%s %s" .Title (.Content | plainify) -}} {{- if partial "is-archive.html" . -}} {{- $archive := partialCached "archive-data.html" . .Path .Language.Lang -}} {{- range $archive.regularPages -}} {{- $text = printf "%s %s %s" $text .Title (partial "archive-preview-html.html" . | plainify) -}} {{- end -}} {{- range $archive.tagSlice -}} {{- $text = printf "%s %s" $text (partial "tag-label.html" .name) -}} {{- end -}} {{- else if and (eq .Section "microblog") .IsSection -}} {{- range partial "microblog-entries.html" . -}} {{- $text = printf "%s %s" $text (.Content | plainify) -}} {{- end -}} {{- end -}} {{- end -}} {{- $needsCJK = gt (len (findRE `\p{Han}` $text 1)) 0 -}} {{- end -}} {{- if $needsCJK -}} {{- $cjkFonts := resources.Get "css/fonts-cjk.css" | resources.Minify | fingerprint -}} {{- end -}}
