{{- /* Tags on the given pages, most used first. */ -}} {{- $tagCounts := dict -}} {{- range . -}} {{- range .Params.tags -}} {{- $name := printf "%v" . -}} {{- $current := index $tagCounts $name | default 0 -}} {{- $tagCounts = merge $tagCounts (dict $name (add $current 1)) -}} {{- end -}} {{- end -}} {{- $tagSlice := slice -}} {{- range $name, $count := $tagCounts -}} {{- $tagSlice = $tagSlice | append (dict "name" $name "count" $count) -}} {{- end -}} {{- return (sort $tagSlice "count" "desc") -}}
