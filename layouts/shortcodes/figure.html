{{- $src := .Get "src" -}} {{- if not $src -}} {{- errorf "figure shortcode requires the 'src' parameter (called at %s)" .Position -}} {{- end -}} {{- $caption := .Get "caption" -}} {{- $title := .Get "title" -}} {{- $alt := .Get "alt" -}} {{- if and (not $alt) $caption -}} {{- $alt = plainify ($caption | markdownify) -}} {{- end -}} {{- if not $alt -}} {{- $alt = "" -}} {{- end -}} {{- $link := .Get "link" -}} {{- $target := .Get "target" -}} {{- $rel := .Get "rel" -}} {{- if and (eq $target "_blank") (eq $rel "") -}} {{- $rel = "noopener" -}} {{- end -}} {{- $resource := .Page.Resources.GetMatch $src -}} {{- $imgURL := $src -}} {{- $imgWidth := 0 -}} {{- $imgHeight := 0 -}} {{- if $resource -}} {{- $imgURL = $resource.RelPermalink -}} {{- $imgWidth = $resource.Width -}} {{- $imgHeight = $resource.Height -}} {{- $subType := lower $resource.MediaType.SubType -}} {{- if and (ne $subType "svg") (ne $subType "gif") -}} {{- $webp := $resource.Process "webp q82" -}} {{- $imgURL = $webp.RelPermalink -}} {{- $imgWidth = $webp.Width -}} {{- $imgHeight = $webp.Height -}} {{- end -}} {{- end -}} {{- if $link -}} {{- end -}} {{- if $link -}} {{- end -}} {{- with $caption -}} {{- end -}}
